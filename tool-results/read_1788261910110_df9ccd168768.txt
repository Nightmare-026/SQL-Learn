    50→    50→function PageRouter({ route, onNavigate }: { route: string; onNavigate: (r: string) => void }) {
    51→    51→  // routes: / /dashboard /module/N /projects /project/ID /search /sandbox /settings
    52→    52→  if (route === '/') return <RootSwitch onNavigate={onNavigate} />;
    53→    53→  if (route === '/dashboard') return <DashboardPage onNavigate={onNavigate} />;
    54→    54→  const moduleMatch = route.match(/^\/module\/(\d+)$/);
    55→    55→  if (moduleMatch) return <ModulePage moduleNumber={Number(moduleMatch[1])} onNavigate={onNavigate} />;
    56→    56→  if (route === '/projects') return <ProjectsPage onNavigate={onNavigate} />;
    57→    57→  const projectMatch = route.match(/^\/project\/([\w-]+)$/);
    58→    58→  if (projectMatch) return <ProjectPage projectId={projectMatch[1]} onNavigate={onNavigate} />;
    59→    59→  if (route === '/search') return <SearchPage onNavigate={onNavigate} />;
    60→    60→  if (route === '/sandbox') return <SandboxPage />;
    61→    61→  if (route === '/settings') return <SettingsPage />;
    62→    62→  return <LandingPage onNavigate={onNavigate} />;
