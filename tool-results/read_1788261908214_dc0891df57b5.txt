    16→// ---------------- Router ----------------
    17→export default function App() {

    50→function PageRouter({ route, onNavigate }: { route: string; onNavigate: (r: string) => void }) {
    51→  // routes: / /dashboard /module/N /projects /project/ID /search /sandbox /settings
    52→  if (route === '/') return <RootSwitch onNavigate={onNavigate} />;
    53→  if (route === '/dashboard') return <DashboardPage onNavigate={onNavigate} />;
    54→  const moduleMatch = route.match(/^\/module\/(\d+)$/);
    55→  if (moduleMatch) return <ModulePage moduleNumber={Number(moduleMatch[1])} onNavigate={onNavigate} />;
    56→  if (route === '/projects') return <ProjectsPage onNavigate={onNavigate} />;
    57→  const projectMatch = route.match(/^\/project\/([\w-]+)$/);
    58→  if (projectMatch) return <ProjectPage projectId={projectMatch[1]} onNavigate={onNavigate} />;
    59→  if (route === '/search') return <SearchPage onNavigate={onNavigate} />;
    60→  if (route === '/sandbox') return <SandboxPage />;
    61→  if (route === '/settings') return <SettingsPage />;
    62→  return <LandingPage onNavigate={onNavigate} />;
    63→}

   378→// ---------------- Dashboard ----------------
   379→function DashboardPage({ onNavigate }: { onNavigate: (r: string) => void }) {
   380→  const t = useT();
   381→  const summary = useProgressSummary();
   382→  const progress = useProgressStore();
   383→  const lang = useLang();
   384→  const [target] = useState(() => nextTargetModule(useProgressStore.getState()));
   385→  const targetEntry = MODULE_INDEX[target];
   386→
   387→  const visibleModules = useMemo(() => {
   388→    const list = [];
   389→    for (let n = 1; n <= 60; n++) {
   390→      if (isModuleUnlocked(n, progress)) {
   391→        list.push(n);
   392→        if (list.length >= 4) break;
   393→      }
   394→    }
   395→    return list;
   396→  }, [progress]);
   397→
   398→  const statusOf = (n: number) => {
   399→    const id = `module-${String(n).padStart(2, '0')}`;
   400→    const mp = progress.modules[id];
   401→    if (mp?.status === 'completed') return 'completed';
   402→    if (mp && (mp.tasksCompleted.length || mp.quizBestScore !== null)) return 'in-progress';
   403→    return 'unlocked';
   404→  };
   405→

   788→function SandboxRunner({ dataset }: { dataset: string }) {
   789→  const t = useT();
   790→  const ctxRef = useRef<DbContext | null>(null);
   791→  const [query, setQuery] = useState('SELECT name FROM sqlite_master WHERE type=\'table\';');
   792→  const [ready, setReady] = useState(false);
   793→  const [runState, setRunState] = useState<{ kind: 'idle' | 'running' | 'result' | 'error'; result?: any; error?: string; ms?: number; rows?: number }>({ kind: 'idle' });
   794→  const textRef = useRef<HTMLTextAreaElement>(null);
   795→  const preRef = useRef<HTMLPreElement>(null);
   796→
   797→  useEffect(() => {
   798→    const ctx = new DbContext(dataset as never);
   799→    ctxRef.current = ctx;
   800→    ctx.ensure().then(() => setReady(true));
   801→  }, [dataset]);
   802→
   803→  const run = async () => {
   804→    if (!ctxRef.current || !query.trim() || runState.kind === 'running') return;
   805→    setRunState({ kind: 'running' });
   806→    const out = await ctxRef.current.run(query);
   807→    if (out.ok) {
   808→      setRunState({ kind: 'result', result: out.result, ms: out.elapsedMs, rows: out.result?.rows.length ?? 0 });
   809→      if (out.mutated) { await ctxRef.current.ensure(); }
   810→    } else {
   811→      setRunState({ kind: 'error', error: out.error });
   812→    }
   813→  };
   814→
   815→  const hlHtml = useMemo(() => {
   816→    const tokens = tokenizeSql(query);
   817→    return tokens.map((tk) => tk.cls ? `<span class="${tk.cls}">${tk.text.replace(/&/g, '&amp;')}</span>` : tk.text).join('');
   818→  }, [query]);
   819→
   820→  if (!ready) return <div className="flex items-center justify-center h-full"><Loader2 className="w-5 h-5 animate-spin text-brand-500" /></div>;
   821→
   822→  return (
   823→    <div className="flex flex-col h-full min-h-0">
   824→      <div className="editor-stack flex-1 min-h-[140px] relative">
   825→        <pre ref={preRef} aria-hidden className="absolute inset-0 p-4 overflow-hidden whitespace-pre-wrap break-words sql-code pointer-events-none" dangerouslySetInnerHTML={{ __html: hlHtml }} />
   826→        <textarea
   827→          ref={textRef}
   828→          value={query}
   829→          onChange={(e) => setQuery(e.target.value)}
   830→          onKeyDown={(e) => { if ((e.ctrlKey || e.metaKey) && e.key === 'Enter') { e.preventDefault(); run(); } }}
   831→          onScroll={() => { if (textRef.current && preRef.current) { preRef.current.scrollTop = textRef.current.scrollTop; } }}
   832→          spellCheck={false}
   833→          className="absolute inset-0 p-4 w-full h-full bg-transparent outline-none resize-none sql-code overflow-auto custom-scroll"
   834→        />
   835→      </div>
   836→      <div className="flex items-center gap-2 px-3 py-2 border-t border-neutral-200 bg-neutral-50/60">
   837→        <button onClick={run} disabled={!query.trim()} className="inline-flex items-center gap-1.5 rounded-lg bg-brand-600 px-4 py-1.5 text-sm font-semibold text-white hover:bg-brand-700 disabled:opacity-40 transition">
   838→          <Play className="w-3.5 h-3.5" /> {t('console.run')}
   839→        </button>
   840→        <button onClick={() => setQuery('')} className="rounded-lg border border-neutral-300 bg-white p-1.5 text-neutral-600 hover:border-neutral-400 transition"><Trash2 className="w-3.5 h-3.5" /></button>
   841→        <button onClick={() => { navigator.clipboard?.writeText(query); toast.success(t('console.copied')); }} className="rounded-lg border border-neutral-300 bg-white p-1.5 text-neutral-600 hover:border-neutral-400 transition"><Copy className="w-3.5 h-3.5" /></button>
   842→        <div className="ml-auto text-xs text-neutral-500">
   843→          {runState.kind === 'result' && `${runState.rows} rows · ${runState.ms?.toFixed(1)}ms`}
   844→        </div>
   845→      </div>
   846→      <div className="flex-1 min-h-[120px] overflow-auto custom-scroll border-t border-neutral-200 p-3 bg-white">
   847→        {runState.kind === 'idle' && <p className="text-sm text-neutral-400 text-center pt-8">{t('console.empty')}</p>}
   848→        {runState.kind === 'running' && <Loader2 className="w-5 h-5 animate-spin text-brand-500 mx-auto mt-8" />}
   849→        {runState.kind === 'error' && (
   850→          <div className="rounded-xl border border-danger-200 bg-danger-50 p-3">
   851→            <div className="font-bold text-danger-700 text-sm mb-1">❌ Error</div>
   852→            <code className="text-xs text-danger-800 sql-code break-words">{runState.error}</code>
   853→          </div>
   854→        )}
   855→        {runState.kind === 'result' && (runState.result
   856→          ? <ResultTable columns={runState.result.columns} rows={runState.result.rows} />
   857→          : <p className="text-sm text-neutral-500 text-center pt-8">{t('console.noResults')}</p>)}
   858→      </div>
   859→    </div>
   860→  );
   861→}
