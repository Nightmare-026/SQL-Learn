     3→// ============ Concept diagrams — REAL-TABLE based (spec §7 diagram field) ============
     4→// Every concept is explained on actual mini database tables:
     5→// header row = columns, data rows = records, with labeled highlights,
     6→// extent arrows (COLUMN / ROW / CELL) and transformation arrows.
     7→// Pedagogical labels are bilingual (EN/HI); table data & SQL keywords stay
     8→// English because they mirror what the learner will actually type.
     9→
    10→import React from 'react';
    11→import type { DiagramKind } from '@/types/content';
    12→import { useLang } from '@/lib/i18n/store';
    13→
    14→const TH = 20; // title bar height
    15→const HH = 24; // header row height
    16→const RH = 22; // data row height
    17→const PAD = 7; // cell text padding
    18→
    19→const C = {
    20→  blue: '#2563EB', blueFill: '#DBEAFE', blueDeep: '#1D4ED8',
    21→  green: '#059669', greenFill: '#D1FAE5',
    22→  amber: '#D97706', amberFill: '#FFEDD5',
    23→  red: '#DC2626', redFill: '#FEE2E2',
    24→  slate: '#64748B', slateFill: '#F1F5F9', slateDeep: '#334155',
    25→  ink: '#0F172A', zebra: '#F1F5F9',
    26→  head: '#1E293B', grid: '#E2E8F0', border: '#94A3B8',
    27→};
    28→const F = 'var(--font-inter), sans-serif';
    29→const M = 'ui-monospace, SFMono-Regular, Menlo, monospace';
    30→
    31→type L = (en: string, hi: string) => string;
    32→
    33→// ---------- geometry ----------
    34→const widths = (cols: Col[]) => cols.map((c) => c.w);
    35→const tblW = (cols: Col[]) => cols.reduce((a, b) => a + b.w, 0);
    36→const colX = (x: number, cols: Col[], i: number) => x + cols.slice(0, i).reduce((a, b) => a + b.w, 0);
    37→const rowY = (y: number, i: number) => y + TH + HH + i * RH;
    38→const rowCY = (y: number, i: number) => rowY(y, i) + RH / 2;
    39→const tblH = (n: number) => TH + HH + n * RH;
    40→
    41→// ---------- mini table ----------
    42→interface Col { label: string; w: number; pk?: boolean; fk?: boolean }
    43→interface TProps {
    44→  x: number; y: number;
    45→  name: string; tag?: string;
    46→  cols: Col[];
    47→  rows: string[][];
    48→  hlCols?: number[];                       // blue column band (the "picked" columns)
    49→  hlRows?: number[];                       // green row band (survivors / current)
    50→  dropRows?: number[];                     // red faded rows (dropped)
    51→  hlCells?: [number, number][];            // amber cell (row ∩ column story)
    52→  tints?: Record<string, string>;          // "r,c" -> custom cell fill
    53→  rowTints?: Record<number, string>;       // custom row fill (partitions / buckets)
    54→  marks?: Record<number, 'pass' | 'drop'>; // ✓ / ✗ circles right of the row
    55→  strike?: number[];                       // strikethrough rows
    56→  partitions?: number[];                   // thick separator above these row indices
    57→  result?: boolean;                        // green title bar (query output)
    58→  fade?: boolean;                          // whole table dimmed (ghost)
    59→}
    60→
    61→function Tbl(p: TProps) {
    62→  const { x, y, cols, rows } = p;
    63→  const W = tblW(cols);
    64→  const H = tblH(rows.length);
    65→  const ws = widths(cols);
    66→  const hlCols = new Set(p.hlCols ?? []);
    67→  const hlRows = new Set(p.hlRows ?? []);
    68→  const dropRows = new Set(p.dropRows ?? []);
    69→  const strike = new Set(p.strike ?? []);
    70→  const hlCells = p.hlCells ?? [];
    71→  const tints = p.tints ?? {};
    72→  const rowTints = p.rowTints ?? {};
    73→  const marks = p.marks ?? {};
    74→  const n = rows.length;
    75→  const accent = p.result ? C.green : C.blue;
    76→
    77→  return (
    78→    <g opacity={p.fade ? 0.5 : 1}>
    79→      {/* zebra striping */}
    80→      {rows.map((_, i) =>
    81→        i % 2 === 1 && !hlRows.has(i) && !dropRows.has(i) && rowTints[i] === undefined ? (
    82→          <rect key={`z${i}`} x={x} y={rowY(y, i)} width={W} height={RH} fill={C.zebra} opacity={0.55} />
    83→        ) : null
    84→      )}
    85→      {/* row bands */}
    86→      {rows.map((_, i) => {
    87→        if (hlRows.has(i)) return <rect key={`rb${i}`} x={x} y={rowY(y, i)} width={W} height={RH} fill={C.greenFill} opacity={0.85} />;
    88→        if (dropRows.has(i)) return <rect key={`rb${i}`} x={x} y={rowY(y, i)} width={W} height={RH} fill={C.redFill} opacity={0.7} />;
    89→        if (rowTints[i] !== undefined) return <rect key={`rb${i}`} x={x} y={rowY(y, i)} width={W} height={RH} fill={rowTints[i]} opacity={0.75} />;
    90→        return null;
    91→      })}
    92→      {/* green left edge for highlighted rows */}
    93→      {[...hlRows].map((i) => (
    94→        <rect key={`lb${i}`} x={x} y={rowY(y, i)} width={3.5} height={RH} fill={C.green} rx={1} />
    95→      ))}
    96→      {/* column bands */}
    97→      {[...hlCols].map((c) => (
    98→        <rect key={`cb${c}`} x={colX(x, cols, c)} y={y + TH + HH} width={cols[c].w} height={n * RH} fill={C.blueFill} opacity={0.6} />
    99→      ))}
   100→      {/* custom cell tints */}
   101→      {Object.entries(tints).map(([k, fill]) => {
   102→        const [r, c] = k.split(',').map(Number);
   103→        return <rect key={`t${k}`} x={colX(x, cols, c)} y={rowY(y, r)} width={cols[c].w} height={RH} fill={fill} opacity={0.9} />;
   104→      })}
   105→      {/* amber hero cells */}
   106→      {hlCells.map(([r, c]) => (
   107→        <rect key={`hc${r}-${c}`} x={colX(x, cols, c)} y={rowY(y, r)} width={cols[c].w} height={RH} fill={C.amberFill} stroke={C.amber} strokeWidth={1.3} />
   108→      ))}
   109→      {/* header row */}
   110→      <rect x={x} y={y + TH} width={W} height={HH} fill={C.head} />
   111→      {cols.map((col, c) =>
   112→        hlCols.has(c) ? <rect key={`hh${c}`} x={colX(x, cols, c)} y={y + TH} width={col.w} height={HH} fill={C.blueFill} /> : null
   113→      )}
   114→      {cols.map((col, c) => (
   115→        <text
   116→          key={`ht${c}`}
   117→          x={colX(x, cols, c) + col.w / 2}
   118→          y={y + TH + HH / 2 + 3.5}
   119→          textAnchor="middle"
   120→          fontSize={9.5}
   121→          fontWeight="700"
   122→          fill={hlCols.has(c) ? C.blueDeep : '#E2E8F0'}
   123→          fontFamily={F}
   124→        >
   125→          {col.pk ? '🔑 ' : col.fk ? '🔗 ' : ''}{col.label}
   126→        </text>
   127→      ))}
   128→      {/* grid lines */}
   129→      {cols.slice(1).map((_, c) => (
   130→        <line key={`v${c}`} x1={colX(x, cols, c + 1)} y1={y + TH + HH} x2={colX(x, cols, c + 1)} y2={y + H} stroke={C.grid} strokeWidth={1} />
   131→      ))}
   132→      {rows.slice(1).map((_, i) => (
   133→        <line key={`h${i}`} x1={x} y1={rowY(y, i + 1)} x2={x + W} y2={rowY(y, i + 1)} stroke={C.grid} strokeWidth={1} />
   134→      ))}
   135→      {(p.partitions ?? []).map((i) => (
   136→        <line key={`p${i}`} x1={x} y1={rowY(y, i)} x2={x + W} y2={rowY(y, i)} stroke={C.slateDeep} strokeWidth={2.2} />
   137→      ))}
   138→      {/* outer border */}
   139→      <rect x={x} y={y} width={W} height={H} rx={6} fill="none" stroke={C.border} strokeWidth={1.4} />
   140→      {/* title bar */}
   141→      <path
   142→        d={`M ${x} ${y + 6} Q ${x} ${y} ${x + 6} ${y} L ${x + W - 6} ${y} Q ${x + W} ${y} ${x + W} ${y + 6} L ${x + W} ${y + TH} L ${x} ${y + TH} Z`}
   143→        fill={accent}
   144→      />
   145→      <text x={x + 9} y={y + 14.5} fontSize={10.5} fontWeight="700" fill="#fff" fontFamily={F}>▦ {p.name}</text>
   146→      {p.tag && <text x={x + W - 8} y={y + 14.5} textAnchor="end" fontSize={8.5} fill="#fff" opacity={0.9} fontFamily={F}>{p.tag}</text>}
   147→      {/* cell text */}
   148→      {rows.map((row, r) =>
   149→        row.map((v, c) => (
   150→          <text
   151→            key={`c${r}-${c}`}
   152→            x={colX(x, cols, c) + PAD}
   153→            y={rowCY(y, r) + 3.5}
   154→            fontSize={9.5}
   155→            fill={dropRows.has(r) || p.fade ? '#94A3B8' : C.ink}
   156→            opacity={dropRows.has(r) ? 0.6 : 1}
   157→            fontFamily={c === 0 && !isNaN(Number(String(v).replace(/[₹,]/g, ''))) ? M : F}
   158→          >
   159→            {v}
   160→          </text>
   161→        ))
   162→      )}
   163→      {/* strikethrough (duplicates) */}
   164→      {[...strike].map((r) => (
   165→        <line key={`s${r}`} x1={x + 4} y1={rowCY(y, r) + 2} x2={x + W - 4} y2={rowCY(y, r) + 2} stroke={C.red} strokeWidth={1.4} opacity={0.8} />
   166→      ))}
   167→      {/* pass / drop marks */}
   168→      {Object.entries(marks).map(([r, m]) => {
   169→        const i = Number(r);
   170→        const pass = m === 'pass';
   171→        return (
   172→          <g key={`m${r}`}>
   173→            <circle cx={x + W + 14} cy={rowCY(y, i)} r={8} fill={pass ? C.greenFill : C.redFill} stroke={pass ? C.green : C.red} strokeWidth={1.3} />
   174→            <text x={x + W + 14} y={rowCY(y, i) + 3.5} textAnchor="middle" fontSize={9} fontWeight="800" fill={pass ? C.green : C.red} fontFamily={F}>
   175→              {pass ? '✓' : '✗'}
   176→            </text>
   177→          </g>
   178→        );
   179→      })}
   180→    </g>
   181→  );
   182→}

   184→// ---------- annotation helpers ----------
   185→function chip(cx: number, cy: number, text: string, fill: string, textColor = '#fff', mono = true) {
   186→  const w = text.length * 5.9 + 18;
   187→  return (
   188→    <g>
   189→      <rect x={cx - w / 2} y={cy - 11} width={w} height={22} rx={11} fill={fill} />
   190→      <text x={cx} y={cy + 3.5} textAnchor="middle" fontSize={10} fontWeight="700" fill={textColor} fontFamily={mono ? M : F}>{text}</text>
   191→    </g>
   192→  );
   193→}
   194→
   195→function txt(x: number, y: number, s: string, o?: { size?: number; color?: string; bold?: boolean; anchor?: 'start' | 'middle' | 'end'; mono?: boolean }) {
   196→  const { size = 10, color = C.ink, bold = false, anchor = 'start', mono = false } = o ?? {};
   197→  return (
   198→    <text x={x} y={y} fontSize={size} fontWeight={bold ? 700 : 400} fill={color} textAnchor={anchor} fontFamily={mono ? M : F}>{s}</text>
   199→  );
   200→}
   201→
   202→function arrow(x1: number, y1: number, x2: number, y2: number, color: keyof typeof C = 'slate', label?: string, labelAbove = true) {
   203→  return (
   204→    <g>
   205→      <line x1={x1} y1={y1} x2={x2} y2={y2} stroke={C[color]} strokeWidth={1.9} markerEnd={`url(#ah-${color})`} />
   206→      {label && (
   207→        <text x={(x1 + x2) / 2} y={(y1 + y2) / 2 + (labelAbove ? -6 : 13)} textAnchor="middle" fontSize={9} fontWeight="600" fill={C[color]} fontFamily={F}>
   208→          {label}
   209→        </text>
   210→      )}
   211→    </g>
   212→  );
   213→}

   215→/** vertical extent indicator ↕ (points at a COLUMN) */
   216→function vExt(x: number, y1: number, y2: number, color = C.blue) {
   217→  return (
   218→    <g>
   219→      <line x1={x} y1={y1 + 5} x2={x} y2={y2 - 5} stroke={color} strokeWidth={1.7} />
   220→      <polygon points={`${x - 4.5},${y1 + 7} ${x + 4.5},${y1 + 7} ${x},${y1}`} fill={color} />
   221→      <polygon points={`${x - 4.5},${y2 - 7} ${x + 4.5},${y2 - 7} ${x},${y2}`} fill={color} />
   222→    </g>
   223→  );
   224→}
   225→
   226→/** horizontal extent indicator ↔ (points at a ROW) */
   227→function hExt(x1: number, x2: number, y: number, color = C.green) {
   228→  return (
   229→    <g>
   230→      <line x1={x1 + 5} y1={y} x2={x2 - 5} y2={y} stroke={color} strokeWidth={1.7} />
   231→      <polygon points={`${x1 + 7},${y - 4.5} ${x1 + 7},${y + 4.5} ${x1},${y}`} fill={color} />
   232→      <polygon points={`${x2 - 7},${y - 4.5} ${x2 - 7},${y + 4.5} ${x2},${y}`} fill={color} />
   233→    </g>
   234→  );
   235→}
   236→
   237→function leader(x1: number, y1: number, x2: number, y2: number, color = C.amber, dashed = true) {
   238→  return <line x1={x1} y1={y1} x2={x2} y2={y2} stroke={color} strokeWidth={1.2} strokeDasharray={dashed ? '3 3' : undefined} />;
   239→}

   241→function defs() {
   242→  return (
   243→    <defs>
   244→      {(['slate', 'blue', 'green', 'amber', 'red'] as const).map((k) => (
   245→        <marker key={k} id={`ah-${k}`} markerWidth="8" markerHeight="6" refX="7" refY="3" orient="auto">
   246→          <polygon points="0 0, 8 3, 0 6" fill={C[k]} />
   247→        </marker>
   248→      ))}
   249→    </defs>
   250→  );
   251→}

   262→export function Diagram({ kind }: { kind: DiagramKind }) {
   263→  const lang = useLang();
   264→  const L: L = (en, hi) => (lang === 'hi' ? hi : en);

   269→        {defs()}
   270→        {render(kind, L)}
   271→      </svg>
   272→    </div>
   273→  );
