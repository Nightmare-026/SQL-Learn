    20→export function ModulePage({ moduleNumber, onNavigate }: { moduleNumber: number; onNavigate: (route: string) => void }) {
    21→  const t = useT();
    22→  const lang = useLang();
    23→  const [module, setModule] = useState<Module | null>(null);
    24→  const [loadError, setLoadError] = useState(false);
    25→  const [tab, setTab] = useState<'theory' | 'quiz' | 'practice' | 'summary'>('theory');
    26→  const [practiceTask, setPracticeTask] = useState(0);
    27→  const [mounted, setMounted] = useState(false);
    28→  const progress = useProgressStore();
    29→  const mp = useProgressStore((s) => s.modules[`module-${String(moduleNumber).padStart(2, '0')}`]);
    30→  const unlocked = isModuleUnlocked(moduleNumber, progress);
    31→  const bumpStats = useProgressStore((s) => s.bumpStats);
    32→  const markTheoryRead = useProgressStore((s) => s.markTheoryRead);
    33→  const recordQuiz = useProgressStore((s) => s.recordQuiz);
    34→  const setLastModule = useProgressStore((s) => s.setLastModule);
    35→  const touch = useProgressStore((s) => s.touch);
    36→
    37→  useEffect(() => {
    38→    setMounted(true);
    39→    let alive = true;
    40→    setLoadError(false);
    41→    setModule(null);
    42→    loadModule(moduleNumber)
    43→      .then((m) => {
    44→        if (!alive) return;
    45→        setModule(m);
    46→        setTab('theory');
    47→        // start at the first unpassed task (moduleNumber captured at effect start)
    48→        const prog = useProgressStore.getState().modules[m.id];
    49→        const doneIds = new Set(prog?.tasksCompleted ?? []);
    50→        const firstOpenIdx = m.tasks.findIndex((tsk) => !doneIds.has(tsk.id));
    51→        setPracticeTask(firstOpenIdx === -1 ? 0 : firstOpenIdx);
    52→      })
    53→      .catch(() => {
    54→        if (alive) setLoadError(true);
    55→      });
    56→    touch();
    57→    setLastModule(`module-${String(moduleNumber).padStart(2, '0')}`);
    58→    return () => { alive = false; };
    59→  }, [moduleNumber, touch, setLastModule]);
    60→
    61→  const entry = MODULE_INDEX[moduleNumber];
    62→  const levelMeta = LEVEL_META[levelOfModule(moduleNumber)];
    63→
    64→  const retryLoad = () => {
    65→    setLoadError(false);
    66→    loadModule(moduleNumber)
    67→      .then((m) => { setModule(m); setTab('theory'); })
    68→      .catch(() => setLoadError(true));
    69→  };
    70→
    71→  if (!mounted) return null;
    72→  if (loadError) {
    73→    return (
    74→      <div className="max-w-xl mx-auto px-4 py-24 text-center">
    75→        <div className="w-14 h-14 rounded-2xl bg-danger-50 flex items-center justify-center mx-auto mb-4">
    76→          <AlertTriangle className="w-7 h-7 text-danger-500" />
    77→        </div>
    78→        <h2 className="font-heading text-xl font-bold text-neutral-800 mb-2">{t('error.moduleLoad.title')}</h2>
    79→        <p className="text-sm text-neutral-600 mb-6">{t('error.moduleLoad.desc')}</p>
    80→        <button onClick={retryLoad} className="inline-flex items-center gap-1.5 rounded-xl bg-brand-600 px-5 py-2 text-sm font-semibold text-white hover:bg-brand-700 transition">
    81→          <RotateCw className="w-4 h-4" /> {t('common.retry')}
    82→        </button>
    83→      </div>
    84→    );
    85→  }
    86→  if (!module) {
    87→    return (
    88→      <div className="max-w-4xl mx-auto px-4 py-24 text-center text-neutral-500">
    89→        <div className="w-6 h-6 border-2 border-brand-500 border-t-transparent rounded-full animate-spin mx-auto mb-4" />
    90→        <p className="text-sm">{t('common.loading')}</p>
    91→      </div>
    92→    );
    93→  }
    94→
    95→  if (!unlocked) {
    96→    return (
    97→      <div className="max-w-xl mx-auto px-4 py-24 text-center">
    98→        <div className="w-16 h-16 rounded-2xl bg-neutral-100 flex items-center justify-center mx-auto mb-4">
    99→          <Lock className="w-7 h-7 text-neutral-400" />
   100→        </div>
   101→        <h2 className="font-heading text-xl font-bold text-neutral-800 mb-2">{t('module.locked.title')}</h2>
   102→        <p className="text-sm text-neutral-600 mb-6">{t('module.locked.desc')}</p>
   103→        <button onClick={() => onNavigate('/')} className="rounded-xl bg-brand-600 px-5 py-2 text-sm font-semibold text-white hover:bg-brand-700 transition">{t('nav.home')}</button>
   104→      </div>
   105→    );
   106→  }
   107→
   108→  const tabs = [
   109→    { id: 'theory', icon: BookOpen, label: t('module.tab.theory') },
   110→    { id: 'quiz', icon: ClipboardList, label: t('module.tab.quiz') },
   111→    { id: 'practice', icon: Terminal, label: t('module.tab.practice') },
   112→    { id: 'summary', icon: CheckCircle2, label: t('module.tab.summary') },
   113→  ] as const;
   114→
   115→  const theoryRead = mp?.theoryRead ?? false;
   116→  const passed = mp?.tasksCompleted ?? [];
   117→  const quizBest = mp?.quizBestScore ?? null;
   118→  const completed = mp?.status === 'completed';
   119→  const nextUnlockedNum = moduleNumber < 60 ? moduleNumber + 1 : null;
   120→
   121→  return (
   122→    <div className="max-w-6xl mx-auto px-3 sm:px-4 py-6">
   123→      {/* Header */}
   124→      <div className="mb-5">
   125→        <div className="flex items-center gap-2 text-xs mb-2">
   126→          <span className={`inline-flex items-center gap-1 rounded-full border px-2 py-0.5 text-[10px] font-bold ${levelMeta.badge}`}>
   127→            <span className="w-1.5 h-1.5 rounded-full" style={{ background: levelMeta.dot }} />
   128→            {t(`level.${entry.level}`)}
   129→          </span>
   130→          <span className="text-neutral-500">M{moduleNumber}</span>
   131→          {completed && <span className="text-success-600 font-bold text-[10px]">✓ {t('status.completed')}</span>}
   132→        </div>
   133→        <h1 className="font-heading text-2xl sm:text-3xl font-bold text-neutral-900 leading-tight mb-1">{module.title[lang]}</h1>
   134→        <div className="flex flex-wrap items-center gap-3 text-xs text-neutral-500">
   135→          <span>⏱ {entry.estimatedTime}</span>
   136→          <span>⌨ {entry.taskCount} {t('stats.tasks').toLowerCase()}</span>
   137→          <span>📝 {module.quiz.length} {t('module.quizCount')}</span>
   138→          <span className="inline-flex items-center gap-1 text-success-700"><BadgeCheck className="w-3.5 h-3.5" />{t('theory.verified')}</span>
   139→        </div>
   140→      </div>
   141→
   142→      {/* Tabs */}
   143→      <div className="flex gap-1 mb-5 border-b border-neutral-200 overflow-x-auto custom-scroll">
   144→        {tabs.map(({ id, icon: Icon, label }) => (
   145→          <button
   146→            key={id}
   147→            onClick={() => setTab(id)}
   148→            className={`inline-flex items-center gap-1.5 px-4 py-2.5 text-sm font-semibold border-b-2 transition whitespace-nowrap ${
   149→              tab === id ? 'border-brand-600 text-brand-700' : 'border-transparent text-neutral-500 hover:text-neutral-800'
   150→            }`}
   151→          >
   152→            <Icon className="w-4 h-4" />
   153→            {label}
   154→            {id === 'practice' && passed.length > 0 && <span className="text-[10px] bg-success-100 text-success-700 rounded-full px-1.5">{passed.length}/{entry.taskCount}</span>}
   155→            {id === 'quiz' && quizBest !== null && <span className="text-[10px] bg-brand-100 text-brand-700 rounded-full px-1.5">{quizBest}%</span>}
   156→          </button>
   157→        ))}
   158→      </div>
   159→
   160→      {/* Content */}
   161→      {tab === 'theory' && (
   162→        <TheoryTab module={module} theoryRead={theoryRead} onRead={() => markTheoryRead(module.id)} />
   163→      )}
   164→      {tab === 'quiz' && (
   165→        <QuizTab module={module} onScore={(pct) => { recordQuiz(module.id, pct); bumpStats({}); }} />
   166→      )}
   167→      {tab === 'practice' && (
   168→        <div className="h-[calc(100vh-260px)] min-h-[520px]">
   169→          <div className="lg:hidden mb-3 rounded-xl border border-warning-200 bg-warning-50 px-4 py-3 text-xs text-warning-800 leading-relaxed">
   170→            <b>⚠ {t('mobile.warn.title')}</b> — {t('mobile.warn.desc')}
   171→          </div>
   172→          <div className="h-full rounded-2xl border border-neutral-200 bg-white shadow-sm overflow-hidden flex flex-col">
   173→            <ConsoleSuspense label={t('common.loading')}>
   174→              <LazyPracticeConsole
   175→              dataset={module.dataset}
   176→              driver={{
   177→                tasks: module.tasks,
   178→                taskIndex: Math.min(practiceTask, module.tasks.length - 1),
   179→                setTaskIndex: (i: number) => setPracticeTask(i),
   180→                onTaskPassed: (taskId) => {
   181→                  markTaskCompleted(module.id, taskId);
   182→                  // auto-advance to the next unpassed task
   183→                  const idx = module.tasks.findIndex((tsk) => tsk.id === taskId);
   184→                  for (let i = idx + 1; i < module.tasks.length; i++) {
   185→                    if (!passed.includes(module.tasks[i].id)) { setPracticeTask(i); return; }
   186→                  }
   187→                },
   188→                onTaskSkipped: (taskId) => markTaskSkipped(module.id, taskId),
   189→                onHintsUsed: (taskId, count) => recordHints(module.id, taskId, count),
   190→                completedTasks: passed,
   191→                skippedTasks: mp?.tasksSkipped ?? [],
   192→                hintsUsed: mp?.hintsUsed ?? {},
   193→                datasetLabel: module.dataset,
   194→              }}
   195→              onQueryRun={() => bumpStats({ totalQueriesRun: 1 })}
   196→            />
   197→            </ConsoleSuspense>
   198→          </div>
   199→        </div>
   200→      )}
   201→      {tab === 'summary' && (
   202→        <SummaryTab module={module} onNavigate={onNavigate} nextUnlocked={nextUnlockedNum} />
   203→      )}
   204→
   205→      {/* Footer nav */}
   206→      <div className="mt-8 flex items-center justify-between gap-2 border-t border-neutral-200 pt-4">
   207→        <button
   208→          onClick={() => onNavigate(moduleNumber > 1 ? `/module/${moduleNumber - 1}` : '/')}
   209→          disabled={moduleNumber === 1}
   210→          className="inline-flex items-center gap-1 rounded-lg border border-neutral-300 bg-white px-3 py-1.5 text-xs font-semibold text-neutral-700 hover:border-neutral-400 disabled:opacity-30 transition"
   211→        >
   212→          <ArrowLeft className="w-3.5 h-3.5" /> {t('module.prev')}
   213→        </button>
   214→        <button onClick={() => onNavigate('/')} className="text-xs text-neutral-500 hover:text-brand-600 inline-flex items-center gap-1 transition">
   215→          <Home className="w-3.5 h-3.5" /> {t('nav.home')}
   216→        </button>
   217→        <button
   218→          onClick={() => nextUnlockedNum && onNavigate(`/module/${nextUnlockedNum}`)}
   219→          disabled={!nextUnlockedNum || !completed}
   220→          className="inline-flex items-center gap-1 rounded-lg bg-brand-600 px-3 py-1.5 text-xs font-semibold text-white hover:bg-brand-700 disabled:opacity-40 transition"
   221→        >
   222→          {t('module.next')} <ArrowRight className="w-3.5 h-3.5" />
   223→        </button>
   224→      </div>
   225→    </div>
   226→  );
   227→
   228→  function markTaskCompleted(moduleId: string, taskId: string) {
   229→    useProgressStore.getState().markTaskCompleted(moduleId, taskId);
   230→  }
   231→  function markTaskSkipped(moduleId: string, taskId: string) {
   232→    useProgressStore.getState().markTaskSkipped(moduleId, taskId);
   233→  }
   234→  function recordHints(moduleId: string, taskId: string, count: number) {
   235→    // `count` is the task's new hint total; only the newly revealed hint counts.
   236→    const prev = useProgressStore.getState().modules[moduleId]?.hintsUsed?.[taskId] ?? 0;
   237→    useProgressStore.getState().recordHints(moduleId, taskId, count);
   238→    useProgressStore.getState().bumpStats({ totalHintsUsed: Math.max(0, count - prev) });
   239→  }
   240→}

   435→// ---------- Reading-progress helpers ----------
   436→function useReadingProgress(containerRef: React.RefObject<HTMLDivElement | null>) {
   437→  const [pct, setPct] = useState(0);
   438→  useEffect(() => {
   439→    let raf = 0;
   440→    const onScroll = () => {
   441→      cancelAnimationFrame(raf);
   442→      raf = requestAnimationFrame(() => {
   443→        const el = containerRef.current;
   444→        if (!el) return;
   445→        const rect = el.getBoundingClientRect();
   446→        const total = rect.height;
   447→        if (total <= 0) return;
   448→        const viewportBottom = window.innerHeight;
   449→        const p = Math.min(1, Math.max(0, (viewportBottom - rect.top) / total));
   450→        setPct(Math.round(p * 100));
   451→      });
   452→    };
   453→    onScroll();
   454→    window.addEventListener('scroll', onScroll, { passive: true });
   455→    window.addEventListener('resize', onScroll);
   456→    return () => { window.removeEventListener('scroll', onScroll); window.removeEventListener('resize', onScroll); cancelAnimationFrame(raf); };
   457→  }, [containerRef]);
   458→  return pct;
   459→}

   461→function useActiveSection(sections: { id: string }[]) {
   462→  const ids = sections.map((s) => s.id);
   463→  const key = ids.join('|');
   464→  const [active, setActive] = useState(ids[0] ?? '');
   465→  useEffect(() => {
   466→    let raf = 0;
   467→    const onScroll = () => {
   468→      cancelAnimationFrame(raf);
   469→      raf = requestAnimationFrame(() => {
   470→        const line = window.scrollY + 160;
   471→        let current = ids[0] ?? '';
   472→        for (const id of ids) {
   473→          const el = document.getElementById(id);
   474→          if (el && el.getBoundingClientRect().top + window.scrollY <= line) current = id;
   475→        }
   476→        setActive(current);
   477→      });
   478→    };
   479→    onScroll();
   480→    window.addEventListener('scroll', onScroll, { passive: true });
   481→    return () => { window.removeEventListener('scroll', onScroll); cancelAnimationFrame(raf); };
   482→  }, [key]);
   483→  return active;
   484→}

   486→function useResumeReading(moduleId: string) {
   487→  // Lazy init from sessionStorage (client-only render — parent gates on `mounted`),
   488→  // so no synchronous setState is needed inside an effect.
   489→  const [resumeY, setResumeY] = useState<number | null>(() => {
   490→    try {
   491→      const saved = Number(sessionStorage.getItem(`theory-scroll:${moduleId}`) ?? '0');
   492→      return saved > 600 ? saved : null;
   493→    } catch { return null; }
   494→  });
   495→  useEffect(() => {
   496→    let raf = 0;
   497→    const onScroll = () => {
   498→      cancelAnimationFrame(raf);
   499→      raf = requestAnimationFrame(() => {
   500→        try { sessionStorage.setItem(`theory-scroll:${moduleId}`, String(window.scrollY)); } catch { /* ignore */ }
   501→      });
   502→    };
   503→    window.addEventListener('scroll', onScroll, { passive: true });
   504→    return () => { window.removeEventListener('scroll', onScroll); cancelAnimationFrame(raf); };
   505→  }, [moduleId]);
   506→  // auto-dismiss the pill after 10s so it never lingers
   507→  useEffect(() => {
   508→    if (resumeY === null) return;
   509→    const timer = setTimeout(() => setResumeY(null), 10000);
   510→    return () => clearTimeout(timer);
   511→  }, [resumeY]);
   512→  const goToResume = () => {
   513→    if (resumeY === null) return;
   514→    setResumeY(null);
   515→    window.scrollTo({ top: resumeY, behavior: 'smooth' });
   516→  };
   517→  return { resumeY, goToResume };
   518→}

   523→/** Subtle scroll-reveal (respects prefers-reduced-motion; content visible even if IO is unavailable). */
   524→function Reveal({ children }: { children: React.ReactNode }) {
   525→  const ref = useRef<HTMLDivElement>(null);
   526→  const skip = PREFERS_REDUCED_MOTION || typeof IntersectionObserver === 'undefined';
   527→  const [shown, setShown] = useState(skip);
   528→  useEffect(() => {
   529→    if (shown) return;
   530→    const el = ref.current;
   531→    if (!el) return;
   532→    const io = new IntersectionObserver((entries) => {
   533→      if (entries.some((e) => e.isIntersecting)) { setShown(true); io.disconnect(); }
   534→    }, { rootMargin: '0px 0px -32px 0px', threshold: 0.03 });
   535→    io.observe(el);
   536→    return () => io.disconnect();
   537→  }, [shown]);
   538→  return (
   539→    <div ref={ref} style={{ willChange: 'opacity, transform' }} className={`transition-[opacity,transform] duration-500 ease-out ${shown ? 'opacity-100 translate-y-0' : 'opacity-0 translate-y-3'}`}>
   540→      {children}
   541→    </div>
   542→  );
   543→}

   656→/** One-shot, dependency-free CSS confetti burst (skipped under prefers-reduced-motion). */
   657→function Confetti({ onceKey }: { onceKey: string }) {
   658→  const [fire, setFire] = useState(false);
   659→  useEffect(() => {
   660→    if (window.matchMedia('(prefers-reduced-motion: reduce)').matches) return;
   661→    let seen = false;
   662→    try { seen = sessionStorage.getItem(onceKey) === '1'; } catch { /* private mode */ }
   663→    if (!seen) {
   664→      setFire(true);
   665→      try { sessionStorage.setItem(onceKey, '1'); } catch { /* ignore */ }
   666→    }
   667→  }, [onceKey]);
   668→  const pieces = useMemo(
   669→    () =>
   670→      Array.from({ length: 64 }, (_, i) => ({
   671→        left: Math.random() * 100,
   672→        delay: Math.random() * 0.9,
   673→        dur: 2.4 + Math.random() * 1.8,
   674→        color: ['#4f46e5', '#22c55e', '#f59e0b', '#ef4444', '#06b6d4', '#a855f7', '#ec4899'][i % 7],
   675→        w: 5 + Math.random() * 5,
   676→        round: Math.random() > 0.6,
   677→      })),
   678→    []
   679→  );
   680→  if (!fire) return null;
   681→  return (
   682→    <div className="pointer-events-none fixed inset-0 z-50 overflow-hidden" aria-hidden="true">
   683→      <style>{`@keyframes sqlconf-fall{0%{transform:translateY(-8vh) rotate(0);opacity:1}100%{transform:translateY(108vh) rotate(660deg);opacity:.5}}`}</style>
   684→      {pieces.map((p, i) => (
   685→        <span
   686→          key={i}
   687→          style={{
   688→            position: 'absolute',
   689→            top: '-4vh',
   690→            left: `${p.left}%`,
   691→            width: p.w,
   692→            height: p.w * (p.round ? 1 : 1.7),
   693→            background: p.color,
   694→            borderRadius: p.round ? '50%' : 2,
   695→            animation: `sqlconf-fall ${p.dur}s ${p.delay}s cubic-bezier(.2,.5,.5,1) forwards`,
   696→          }}
   697→        />
   698→      ))}
   699→    </div>
   700→  );
   701→}
